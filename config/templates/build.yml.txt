name: Build plugin
on:
  push:
  workflow_dispatch:
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-dotnet@v4
        with:
          dotnet-version: '10.0.x'
      - run: dotnet build src/{{id}}.Plugin.csproj -c Release
      - uses: actions/upload-artifact@v4
        with:
          name: {{id}}-plugin
          path: src/bin/Release/{{targetFramework}}/*.dll
